cmake_minimum_required(VERSION 3.20)
project(tp_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(tp_core STATIC app/controller.c app/preflight.c app/health.c app/event_transition.c app/clock_cache.c app/valve_policy.c app/pulse_counter.c app/heater_duty.c app/measurement.c app/schedule.c app/service.c app/door.c app/door_service.c
    drivers/ads1220.c drivers/adc_reader.c drivers/mb85rc16.c drivers/ds3231.c drivers/ntc.c drivers/pn532.c drivers/pn532_scan.c protocol/modbus_rtu.c protocol/rtu_receiver.c protocol/rtu_port.c protocol/subordinate.c protocol/subordinate_poll.c protocol/subordinate_link.c
    storage/config_store.c storage/event_log.c bootloader/boot.c bootloader/download.c bootloader/upgrade.c)
target_include_directories(tp_core PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(tp_core PRIVATE -Wall -Wextra -Werror)
endif()
if(NOT MSVC)
    target_link_libraries(tp_core PUBLIC m)
endif()
option(TP_BUILD_HOST_TESTS "Build host-side checks; this does not build STM32 firmware" ON)
if(TP_BUILD_HOST_TESTS)
    enable_testing()
    add_executable(tp_simulator examples/simulator.c)
    target_link_libraries(tp_simulator PRIVATE tp_core)
    add_test(NAME simulator_normal COMMAND tp_simulator)
    add_test(NAME simulator_interlock COMMAND tp_simulator interlock)
    add_test(NAME simulator_sensor COMMAND tp_simulator sensor)
    foreach(name door door_service pn532 pn532_scan subordinate subordinate_poll subordinate_link upgrade download boot preflight health log_service event_transition event_log integration ads_schedule service rtu_receiver fram settings controlled rtu_port ds3231 clock_cache valves pulse_counter heater_duty ntc)
        add_executable(test_${name} tests/test_${name}.c)
        target_link_libraries(test_${name} PRIVATE tp_core)
        add_test(NAME ${name} COMMAND test_${name})
    endforeach()
endif()
